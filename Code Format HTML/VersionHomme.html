calcul_prime <- function(taux_mortalite, ages, K = 20000, taux_technique = 0.02) {

  n <- length(taux_mortalite)  
  v <- 1 / (1 + taux_technique)  
  
  p_x <- 1 - taux_mortalite  
  t_p_x <- cumprod(c(1, p_x))[-length(p_x)-1]  
  t_q_x <- t_p_x * taux_mortalite  
  
  vap_benefice <- sum(K * t_q_x * v^(1:n))
  
  vap_prime <- sum(t_p_x * v^(0:(n-1)))
  
  prime <- vap_benefice / vap_prime
  
  return(prime)
}
